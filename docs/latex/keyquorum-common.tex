% the float package's placement machinery -- these are meant to sit right

\newcommand{\Screenshot}[2]{%
  \par\smallskip\noindent
  \fbox{\includegraphics[width=\dimexpr0.94\linewidth-2\fboxsep-2\fboxrule\relax,height=0.4\textheight,keepaspectratio]{#1}}\par
  {\small\color{KqSlate} #2}\par\smallskip
}

% Short framed callouts for notes, warnings, and security-critical points.
\newsavebox{\KqCalloutBox}
\newenvironment{KqCallout}[2]{%
  \def\KqCalloutColor{#1}%
  \setlength{\fboxrule}{0.6pt}%
  \par\smallskip\noindent
  \begin{lrbox}{\KqCalloutBox}%
  \begin{minipage}{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}%
  \setlength{\parskip}{3pt}%
  {\bfseries\color{#1}#2}\par
}{%
  \end{minipage}%
  \end{lrbox}%
  \fcolorbox{\KqCalloutColor}{\KqCalloutColor!6}{\usebox{\KqCalloutBox}}%
  \par\smallskip
}
\newenvironment{noteBox}[1][Note]{\begin{KqCallout}{KqTeal}{#1}}{\end{KqCallout}}
\newenvironment{cautionBox}[1][Watch out]{\begin{KqCallout}{KqGold}{#1}}{\end{KqCallout}}
\newenvironment{securityBox}[1][Security]{\begin{KqCallout}{KqRed}{#1}}{\end{KqCallout}}
